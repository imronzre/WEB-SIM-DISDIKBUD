import { NextResponse, type NextRequest } from 'next/server';
import { createServerClient, type CookieOptions } from '@supabase/ssr';

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({
    request: {
      headers: request.headers,
    },
  });

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
  const anonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || '';

  const supabase = createServerClient(url, anonKey, {
    cookies: {
      getAll() {
        return request.cookies.getAll();
      },
      setAll(cookiesToSet: { name: string; value: string; options: CookieOptions }[]) {
        cookiesToSet.forEach(({ name, value }) =>
          request.cookies.set(name, value)
        );
        response = NextResponse.next({
          request: {
            headers: request.headers,
          },
        });
        cookiesToSet.forEach(({ name, value, options }) =>
          response.cookies.set(name, value, options)
        );
      },
    },
  });

  // Verifikasi sesi pengguna aktif via Supabase Auth
  const {
    data: { user },
  } = await supabase.auth.getUser();

  const pathname = request.nextUrl.pathname;

  // Daftar rute yang wajib dilindungi dengan autentikasi
  const protectedRoutes = [
    '/dashboard',
    '/siswa',
    '/lembaga',
    '/kecamatan',
    '/laporan',
    '/unggah',
    '/pengguna',
    '/log-aktivitas',
  ];

  const isProtected = protectedRoutes.some(
    (route) => pathname === route || pathname.startsWith(`${route}/`)
  );
  const isRoot = pathname === '/';
  const isLogin = pathname === '/login';

  // 1. Jika pengguna membuka halaman utama (/) atau rute terlindungi tanpa sesi login aktif -> REDIRECT KE /login
  if (!user && (isProtected || isRoot)) {
    const redirectUrl = request.nextUrl.clone();
    redirectUrl.pathname = '/login';
    return NextResponse.redirect(redirectUrl);
  }

  // 2. Jika pengguna sudah login dan mencoba mengakses /login atau / -> REDIRECT KE /dashboard
  if (user && (isLogin || isRoot)) {
    const redirectUrl = request.nextUrl.clone();
    redirectUrl.pathname = '/dashboard';
    return NextResponse.redirect(redirectUrl);
  }

  return response;
}

    const jenjang = searchParams.get('jenjang');

    const supabase = await createServerSupabaseClient();
    let query = supabase
      .from('lembaga')
      .select(`
        *,
        kecamatan:kecamatan_id (nama)
      `)
      .order('nama', { ascending: true });

    if (kecamatanId && kecamatanId !== 'all') {
      query = query.eq('kecamatan_id', kecamatanId);
    }
    if (jenjang && jenjang !== 'all') {
      query = query.eq('jenjang', jenjang);

    if (q) {
      query = query.ilike('nama', `%${q}%`);
    }

    const { data, error } = await query;
    if (error) {
      return NextResponse.json({ error: error.message }, { status: 500 });
    }

    const formatted = (data || []).map((l: any) => ({

      kecamatan_nama: l.kecamatan?.nama || '',
    }));

    return NextResponse.json({ data: formatted });

import { NextRequest, NextResponse } from 'next/server';
import { createServerSupabaseClient } from '@/lib/supabase/server';

export async function POST(req: NextRequest) {
  try {
    const { validRows, invalidCount, lembagaId, mode, fileName, user } = await req.json();

    if (!lembagaId) {
      return NextResponse.json({ error: 'Lembaga ID diperlukan' }, { status: 400 });
    }

    const supabase = await createServerSupabaseClient();

    // 1. Jika mode = 'ganti_semua', hapus data lama
    if (mode === 'ganti_semua') {
      await supabase.from('siswa').delete().eq('lembaga_id', lembagaId);
    }

    // 2. Ambil siswa yang sudah ada untuk matching
    const { data: existingSiswa } = await supabase
      .from('siswa')
      .select('id, nisn, nama, tanggal_lahir')
      .eq('lembaga_id', lembagaId);

    const existing = existingSiswa || [];
    let inserted = 0;
    let updated = 0;

    for (const row of validRows) {
      let matchId: string | null = null;

      if (row.nisn) {
        const found = existing.find((s) => s.nisn === row.nisn);
        if (found) matchId = found.id;
      } else {
        const found = existing.find(
          (s) =>
            s.nama.toLowerCase().trim() === row.nama.toLowerCase().trim() &&
            s.tanggal_lahir === row.tanggal_lahir
        );
        if (found) matchId = found.id;
      }

      if (matchId && mode !== 'ganti_semua') {
        // Update
        await supabase
          .from('siswa')
          .update({
            nisn: row.nisn || null,
            nama: row.nama,
            tanggal_lahir: row.tanggal_lahir,
            alamat: row.alamat,
            nama_ayah: row.nama_ayah,
            nama_ibu: row.nama_ibu,
            updated_at: new Date().toISOString(),
          })
          .eq('id', matchId);
        updated++;
      } else {
        // Insert
        await supabase.from('siswa').insert({
          nisn: row.nisn || null,
          nama: row.nama,
          tanggal_lahir: row.tanggal_lahir,
          alamat: row.alamat,
          nama_ayah: row.nama_ayah,
          nama_ibu: row.nama_ibu,
          lembaga_id: lembagaId,
        });
        inserted++;
      }
    }

    // 3. Catat upload_log
    await supabase.from('upload_log').insert({
      lembaga_id: lembagaId,
      user_id: user?.id || null,
      user_nama: user?.nama || 'Petugas',
      nama_file: fileName,
      mode: mode || 'perbarui',
      baris_berhasil: validRows.length,
      baris_gagal: invalidCount || 0,
    });

    // 4. Catat activity_log
    await supabase.from('activity_log').insert({
      user_id: user?.id || null,
      user_nama: user?.nama || 'Petugas',
      user_role: user?.role || 'lembaga',
      aksi: 'UNGGAH_DATA',
      detail: `Unggah berkas ${fileName}: ${validRows.length} berhasil, ${invalidCount || 0} gagal [Mode: ${mode}]`,
    });

    return NextResponse.json({
      success: true,
      successCount: validRows.length,
      insertedCount: inserted,
      updatedCount: updated,
    });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

/**
 * Parse tanggal dari berbagai format (string DD-MM-YYYY, YYYY-MM-DD, atau nomor serial Excel)
 */
export function parseDateString(raw: any): { isoDate: string; displayDate: string; error?: string } {
  if (!raw) return { isoDate: '', displayDate: '', error: 'Tanggal lahir wajib diisi' };

  let d: Date | null = null;

  if (typeof raw === 'number') {
    // Excel serial number
    const parsed = XLSX.SSF.parse_date_code(raw);
    if (parsed) {
      d = new Date(Date.UTC(parsed.y, parsed.m - 1, parsed.d));
    }
  } else if (typeof raw === 'string') {
    const trimmed = raw.trim();
    // Format DD-MM-YYYY atau DD/MM/YYYY
    const dmyMatch = trimmed.match(/^(\d{1,2})[-/](\d{1,2})[-/](\d{4})$/);
    if (dmyMatch) {
      const day = parseInt(dmyMatch[1], 10);
      const month = parseInt(dmyMatch[2], 10) - 1;
      const year = parseInt(dmyMatch[3], 10);
      d = new Date(Date.UTC(year, month, day));
    } else {
      // Format YYYY-MM-DD
      const ymdMatch = trimmed.match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})$/);
      if (ymdMatch) {
        const year = parseInt(ymdMatch[1], 10);
        const month = parseInt(ymdMatch[2], 10) - 1;
        const day = parseInt(ymdMatch[3], 10);
        d = new Date(Date.UTC(year, month, day));
      } else {
        const fallback = new Date(trimmed);
        if (!isNaN(fallback.getTime())) {
          d = fallback;
        }
      }
    }
  } else if (raw instanceof Date) {
    d = raw;
  }

  if (!d || isNaN(d.getTime())) {
    return {
      isoDate: '',
      displayDate: String(raw),
      error: `Format tanggal tidak valid (${raw}). Gunakan DD-MM-YYYY (contoh: 15-05-2019)`,
    };
  }

  const now = new Date();
  if (d.getTime() > now.getTime()) {
    return {
      isoDate: '',
      displayDate: String(raw),
      error: 'Tanggal lahir tidak boleh di masa depan',
    };
  }

  const year = d.getUTCFullYear();
  const month = String(d.getUTCMonth() + 1).padStart(2, '0');
  const day = String(d.getUTCDate()).padStart(2, '0');
  const isoDate = `${year}-${month}-${day}`;
  const displayDate = `${day}-${month}-${year}`;

  return { isoDate, displayDate };
}

/**
 * Validasi dan Parsing Array Buffer File Excel
 */
export function parseAndValidateExcel(buffer: ArrayBuffer): {
  rows: ParsedExcelRow[];
  totalRows: number;
  validRowsCount: number;
  invalidRowsCount: number;
} {
  const wb = XLSX.read(buffer, { type: 'array' });
  const firstSheetName = wb.SheetNames[0];
  const ws = wb.Sheets[firstSheetName];

  // Convert to JSON array of objects
  const rawRows: Record<string, any>[] = XLSX.utils.sheet_to_json(ws, { defval: '' });

  const parsedRows: ParsedExcelRow[] = [];
  const seenKeys = new Map<string, number>();

  rawRows.forEach((row, idx) => {
    const rowNumber = idx + 2; // baris di Excel (header = 1)
    const errors: string[] = [];
    const warnings: string[] = [];

    // Mencari kunci kolom fleksibel (case-insensitive)
    const findVal = (keywords: string[]): string => {
      for (const k of Object.keys(row)) {
        const cleanK = k.toLowerCase().trim();
        for (const kw of keywords) {
          if (cleanK.includes(kw)) {
            return String(row[k] ?? '').trim();
          }
        }
      }
      return '';
    };

    const nisnRaw = findVal(['nisn']);
    const nama = findVal(['nama peserta', 'nama siswa', 'nama']);
    const tglRaw = findVal(['tanggal lahir', 'tgl lahir', 'tanggal', 'tgl']);
    const alamat = findVal(['alamat']);
    const namaAyah = findVal(['ayah', 'nama ayah']);
    const namaIbu = findVal(['ibu', 'nama ibu']);

    // Validasi NISN jika diisi
    let cleanNisn = nisnRaw.replace(/[^0-9]/g, '');
    if (nisnRaw && cleanNisn.length !== 10) {
      errors.push(`NISN harus 10 digit angka (ditemukan: "${nisnRaw}")`);
    }

    // Validasi Nama Wajib
    if (!nama) {
      errors.push('Nama peserta didik wajib diisi');
    }

    // Validasi Tanggal Lahir
    const dateParsed = parseDateString(tglRaw);
    if (dateParsed.error) {
      errors.push(dateParsed.error);
    }

    // Validasi Alamat Wajib
    if (!alamat) {
      errors.push('Alamat wajib diisi');
    }

    // Validasi Nama Ayah Wajib
    if (!namaAyah) {
      errors.push('Nama Ayah wajib diisi');
    }

    // Validasi Nama Ibu Wajib
    if (!namaIbu) {
      errors.push('Nama Ibu wajib diisi');
    }

    // Deteksi duplikat dalam file
    const uniqueKey = cleanNisn
      ? `NISN:${cleanNisn}`
      : `NAME_DOB:${nama.toLowerCase()}_${dateParsed.isoDate}`;

    if (nama && dateParsed.isoDate) {
      if (seenKeys.has(uniqueKey)) {
        errors.push(`Duplikat di baris ${seenKeys.get(uniqueKey)} dalam file yang sama`);
      } else {
        seenKeys.set(uniqueKey, rowNumber);
      }
    }

    parsedRows.push({
      rowNumber,
      nisn: cleanNisn,
      nama,
      tanggal_lahir: dateParsed.isoDate,
      alamat,
      nama_ayah: namaAyah,
      nama_ibu: namaIbu,
      isValid: errors.length === 0,
      errors,
      warnings,
    });
  });

  const validRowsCount = parsedRows.filter((r) => r.isValid).length;
  const invalidRowsCount = parsedRows.filter((r) => !r.isValid).length;

  return {
    rows: parsedRows,
    totalRows: parsedRows.length,
    validRowsCount,
    invalidRowsCount,
  };
}

  static bulkUpsertSiswa(
    validRows: ParsedExcelRow[],
    invalidCount: number,
    lembagaId: string,
    mode: 'perbarui' | 'ganti_semua',
    fileName: string,
    user: UserProfile
  ): { successCount: number; updatedCount: number; insertedCount: number } {
    let list = loadFromStorage<Siswa[]>('siswa', INITIAL_SISWA);
    const lem = this.getLembagaById(lembagaId);
    if (!lem) throw new Error('Lembaga tidak ditemukan');

    let insertedCount = 0;
    let updatedCount = 0;

    if (mode === 'ganti_semua') {
      list = list.filter((s) => s.lembaga_id !== lembagaId);
    }

    validRows.forEach((row) => {
      let existingIdx = -1;
      if (row.nisn) {
        existingIdx = list.findIndex(
          (s) => s.lembaga_id === lembagaId && s.nisn === row.nisn
        );
      } else {
        existingIdx = list.findIndex(
          (s) =>
            s.lembaga_id === lembagaId &&
            s.nama.toLowerCase().trim() === row.nama.toLowerCase().trim() &&
            s.tanggal_lahir === row.tanggal_lahir
        );
      }

      if (existingIdx !== -1 && mode !== 'ganti_semua') {
        list[existingIdx] = {
          ...list[existingIdx],
          nisn: row.nisn || null,
          nama: row.nama,
          tanggal_lahir: row.tanggal_lahir,
          alamat: row.alamat,
          nama_ayah: row.nama_ayah,
          nama_ibu: row.nama_ibu,
          updated_at: new Date().toISOString(),
        };
        updatedCount++;
      } else {
        list.push({
          id: `sis-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`,
          nisn: row.nisn || null,
          nama: row.nama,
          tanggal_lahir: row.tanggal_lahir,
          alamat: row.alamat,
          nama_ayah: row.nama_ayah,

          jenjang: lem.jenjang,
          kecamatan_id: lem.kecamatan_id,
          kecamatan_nama: lem.kecamatan_nama,
          created_at: new Date().toISOString(),
          updated_at: new Date().toISOString(),
        });
        insertedCount++;
      }
    });

    saveToStorage('siswa', list);

    const lemList = loadFromStorage<Lembaga[]>('lembaga', INITIAL_LEMBAGA);
    const targetLemIdx = lemList.findIndex((l) => l.id === lembagaId);
    if (targetLemIdx !== -1) {
      lemList[targetLemIdx].last_upload = new Date().toISOString();
      saveToStorage('lembaga', lemList);
    }

    const uploadLogs = loadFromStorage<UploadLog[]>('upload_logs', []);
    const newUploadLog: UploadLog = {
      id: `up-${Date.now()}`,
      lembaga_id: lem.id,
      lembaga_nama: lem.nama,
      user_id: user.id,
      user_nama: user.nama,
      nama_file: fileName,
      mode,
      baris_berhasil: validRows.length,
      baris_gagal: invalidCount,
      created_at: new Date().toISOString(),
    };
    uploadLogs.unshift(newUploadLog);
    saveToStorage('upload_logs', uploadLogs);

    this.addActivityLog(
      'UNGGAH_DATA',
      `Unggah file ${fileName} (${validRows.length} berhasil, ${invalidCount} gagal) untuk ${lem.nama} [Mode: ${mode}]`,
      user
    );

    // Call server-side API bulk route if online
    if (typeof window !== 'undefined') {
      fetch('/api/siswa/bulk', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          validRows,
          invalidCount,
          lembagaId,
          mode,
          fileName,
          user,
        }),
      }).catch((e) => console.warn('Background bulk API sync:', e));
    }

    return {
      successCount: validRows.length,
      insertedCount,
      updatedCount,
    };
  }

export function formatDateIndo(dateStr: string | null | undefined): string {
  if (!dateStr) return '-';
  try {
    const d = new Date(dateStr);
    if (isNaN(d.getTime())) return dateStr;
    return new Intl.DateTimeFormat('id-ID', {
      day: 'numeric',
      month: 'long',
      year: 'numeric',
    }).format(d);
  } catch {
    return dateStr;
  }
}

      month: 'short',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
    }).format(d);
  } catch {
    return dateStr;

export function calculateAge(birthDateStr: string): number | null {
  if (!birthDateStr) return null;
  const birth = new Date(birthDateStr);
  if (isNaN(birth.getTime())) return null;
  const today = new Date();
  let age = today.getFullYear() - birth.getFullYear();
  const m = today.getMonth() - birth.getMonth();
  if (m < 0 || (m === 0 && today.getDate() < birth.getDate())) {
    age--;
  }
  return age >= 0 ? age : null;
}
